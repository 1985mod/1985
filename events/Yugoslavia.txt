add_namespace = yugoslavia

### Единый ивент: Распад Югославии
country_event = {
	id = yugoslavia.1
	title = yugoslavia.1.t
	desc = yugoslavia.1.desc
	hidden = yes

	trigger = {
		tag = YUG
		has_civil_war = no
		date > 1991.06.25
	}
	immediate = {
		# Словения – 25 июня 1991
		if = {
			limit = {
				date > 1991.06.25
				NOT = { has_country_flag = collapse_SLV }
			}
			release = SLV
			set_country_flag = collapse_SLV
		}

		# Хорватия – 25 июня 1991
		if = {
			limit = {
				date > 1991.06.25
				NOT = { has_country_flag = collapse_CRO }
			}
            release = CRO
			set_country_flag = collapse_CRO
		}

		# Македония – 8 сентября 1991
		if = {
			limit = {
				date > 1991.09.08
				NOT = { has_country_flag = collapse_MAC }
			}
			MAC = { transfer_state = 106 }
			MAC = { transfer_state = 970 }
			set_country_flag = collapse_MAC
		}

		# Босния и Герцеговина – 1 марта 1992
		if = {
			limit = {
				date > 1992.03.01
				NOT = { has_country_flag = collapse_BOS }
			}
			BOS = { transfer_state = 104 }
			BOS = { transfer_state = 804 }
			BOS = { transfer_state = 1048 }
			BOS = { transfer_state = 1052 }
			BOS = { transfer_state = 1053 }
			set_country_flag = collapse_BOS
		}

		# Черногория – 3 июня 2006
		if = {
			limit = {
				date > 2006.06.03
				NOT = { has_country_flag = collapse_MNT }
			}
			MNT = { transfer_state = 105 }
			set_country_flag = collapse_MNT

			SER = { transfer_state = 107 }
			SER = { transfer_state = 45 }
			SER = { transfer_state = 764 }
			SER = { transfer_state = 108 }
			SER = { transfer_state = 802 }
			SER = { transfer_state = 803 }
		}

		# Косово – 17 февраля 2008
		if = {
			limit = {
				date > 2008.02.17
				NOT = { has_country_flag = collapse_KOS }
			}
			KOS = { transfer_state = 802 }
			set_country_flag = collapse_KOS
		}
	}
}
