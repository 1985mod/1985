country_event = {   ### Ирак вторгается в Кувейт ###
    id = iraq.2
    title = iraq.2.t
    desc = iraq.2.d
    picture = GFX_report_event_generic_peace

    trigger = {
        tag = IRQ
        has_country_leader = { character = IRQ_Saddam_Hussein ruling_only = yes }
        date > 1990.8.1

    fire_only_once = yes

    option = {
        name = iraq.2.a
        set_global_flag = iraq_invade_in_kuwait

            diplomatic_relation = {

                active = no
            }
        }

            diplomatic_relation = {
                country = KUW
                relation = guarantee
                active = no
            }
        }
        UAE = {
            diplomatic_relation = {
                country = KUW
                relation = guarantee
                active = no
            }
        }
        OMA = {
            diplomatic_relation = {
                country = KUW
                relation = guarantee
                active = no
            }
        }
    }

    option = {
        name = iraq.2.b
        add_stability = -0.05

    }
}
