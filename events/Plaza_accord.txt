    option = {
        name = plaza_accord.5.a

		}

			limit = {

			}

			remove_ideas = USA_trade_deficit
			add_timed_idea = {

			}

			limit = {
				OR = {
					tag = WGR

				idea = WGR_plaza_accord

			}
		}

			limit = {
				tag = JAP
			}

			}
		}
    }
}
