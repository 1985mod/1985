country_event = {
    id = SOV_collapse.1
    title = SOV_collapse.1.t
    desc = SOV_collapse.1.desc
    hidden = yes

    trigger = {
        tag = SOV
        date > 1990.03.11
		has_country_flag = gorbachev
    }

    immediate = {
        # Выход из ОВД (Польша, Чехия, ГДР, Венгрия, Румыния, Болгария)
        if = {
            limit = {
                date > 1990.03.11
                NOT = { has_country_flag = collapse_OVD }
            }
			
            set_country_flag = collapse_right_now

			every_country = {
			    limit = { is_in_faction_with = SOV }
			    leave_faction = yes
			}

			set_autonomy = { 
				target = POL
				autonomy_state = autonomy_free
			}
			set_autonomy = {
				target = CZE
				autonomy_state = autonomy_free
			}
			set_autonomy = {
				target = DDR
				autonomy_state = autonomy_free
			}
			set_autonomy = { 
				target = HUN
				autonomy_state = autonomy_free
			}
			set_autonomy = {
				target = ROM
				autonomy_state = autonomy_free
			}
			set_autonomy = {
				target = BUL
				autonomy_state = autonomy_free
			}

            set_country_flag = collapse_OVD
        }

        # Прибалтика — Литва (март 1990)
        if = {
            limit = {
                date > 1990.03.11
                NOT = { has_country_flag = collapse_LIT }
            }
            release = LIT
