country_event = {
	id = switzerland.1
	title = switzerland.1.t
	desc = switzerland.1.desc
	hidden = yes

	trigger = {
		tag = SWI
		has_civil_war = no
		date > 1985.12.14
	}

	fire_only_once = yes

	option = {
		name = switzerland.1.a
		effect_tooltip = {
			set_politics = {
				ruling_party = conservatism
				elections_allowed = yes
			}
		}

		hidden_effect = {
			retire_ideology_leader = conservatism
			promote_character = {
				character = SWI_alphons_egli
				ideology = conservatism
			}
			set_politics = {
				ruling_party = conservatism
				elections_allowed = yes
			}
		}
	}
}
