add_namespace = KOR_unification

country_event = {
	id = KOR_unification.1
	title = KOR_unification.1.t
	desc = KOR_unification.1.desc

	trigger = {
		OR = {
			AND = {
				tag = KOR
				OR = {
					has_government = communism
					controls_state = 525
				}
			}
			AND = {
				tag = PRK
				OR = {
					has_government = liberalism
					has_government = conservatism
					controls_state = 527
				}
			}
			
		}
	}

	hidden = yes

	immediate = {
		if = {
			limit = {
				tag = KOR
				has_government = communism
			}
			PRK = {
				annex_country = {
					target = KOR
					transfer_troops = no
				}
			}
		}
		if = {
			limit = {
				tag = PRK
				OR = {
					has_government = socialism
					has_government = liberalism
				}
			}
			KOR = {
				annex_country = {
					target = PRK
					transfer_troops = no
				}
			}
		}
	}
}
