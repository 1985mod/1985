country_event = {
	id = DDR_unification.1
	title = DDR_unification.1.t
	desc = DDR_unification.1.desc

	trigger = {
		tag = DDR
		NOT = { has_global_flag = GER_unification1 }
		is_in_faction = no
		is_puppet = no
		date > 1990.10.3 
		}
		set_politics = { 
            ruling_party = communism
		}
	}

	option = { # ГДР согласна
		name = DDR_unification.1.a
		ai_chance = {
			base = 20
		}

		effect_tooltip = {
			WGR = { 
				annex_country = {
					target = DDR
				}
			}			
		}

		hidden_effect = {
			set_global_flag = GER_unification1
			WGR = { transfer_state = 60 }
			WGR = { transfer_state = 61 }
			WGR = { transfer_state = 62 }
			WGR = { transfer_state = 64 }
			WGR = { transfer_state = 65 }
			WGR = {
				add_core_of = 60
				add_core_of = 61
				add_core_of = 62
				add_core_of = 64
				add_core_of = 65
			}
			WGR = { capital = 64 }
		}
	}

	option = { # ГДР отказывается
		name = DDR_unification.1.b
		ai_chance = {
			base = 10
		}

		effect_tooltip = {
		}

		hidden_effect = {
			set_global_flag = GER_unification1
			WGR = {
				country_event = {
					id = DDR_unification.2
					days = 1
				}
			}
		}
	}
}
