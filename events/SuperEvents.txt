add_namespace = super_event

country_event = {
	id = super_event.1
	title = super_event.1.t
	desc = super_event.1.d
	picture = GFX_report_event_diplomatic_message

	is_triggered_only = no
	
	fire_only_once = yes
	
	hidden = yes
	
	trigger = {
		AND = {
			SOV = { has_government = communism }
			OR = {
				SOV = { has_war_with = USA }
				SOV = { has_war_with = ENG }
			}
			NOT = {
				has_global_flag = super_world_war_3
			}
		}
	}

	option = {
		name = super.1.a
		
        set_global_flag = sov_world_war_3
		set_global_flag = super_event_visible
		play_song = "old_march_name"
		
		set_global_flag = super_world_war_3
	}
}
