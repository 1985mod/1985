add_namespace = amn

country_event = {
    id = amn.1
    title = amn.1.t
    desc = amn.1.desc

	trigger = {
        AFS = {
            NOT = { has_war_with = AFN }
        }
        tag = AFS
        country_exists = AFS
    }
	fire_only_once = yes

    immediate = {
		promote_character = {
			character = AFS_sibghatullah_mojaddedi
			ideology = neutrality
		}
        set_country_flag = {
            flag = sibghatullah_mojaddedi_country_leader
            days = 60
            value = 1
        }
    }

    option = {
        name = amn.1.a
        set_cosmetic_tag = AFS_islamic_state_of_afghanistan
        add_stability = 0.03
		set_politics = {
			ruling_party = neutrality
			elections_allowed = no
		}
		add_popularity = {
			ideology = neutrality
			popularity = 0.1
		}
    }
}
