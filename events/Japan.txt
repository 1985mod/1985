add_namespace = japan

country_event = {
	id = japan.1
	title = japan.1.t
	desc = japan.1.desc
	hidden = yes

	trigger = {
		tag =  JAP
		has_civil_war = no
		date > 1987.11.6
	}

	fire_only_once = yes

	option = {
		name = japan.1.a
		effect_tooltip = {
			set_politics = {
				ruling_party = conservatism
				elections_allowed = yes
			}
		}

		hidden_effect = {
			retire_ideology_leader = conservatism
			promote_character = {
				character = JAP_noboru_takeshita
				ideology = conservatism
			}
			set_politics = {
				ruling_party = conservatism
				elections_allowed = yes
			}
		}
	}
}
