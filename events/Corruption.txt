add_namespace = corruption

country_event = {
    id = corruption.1
    title = corruption.1.t
    desc = corruption.1.desc
	picture = GFX_report_generic_corruption

	is_triggered_only = yes

	immediate = {		
		hidden_effect = {
			if = {
				limit = {
					has_idea = significant_corruption
				}
				swap_ideas = {
					remove_idea = significant_corruption
					add_idea = horrific_corruption
				}  
			}
			if = {
				limit = {
					has_idea = moderate_corruption
				}
				swap_ideas = {
					remove_idea = moderate_corruption
					add_idea = significant_corruption
				}  
			}	
			if = {
				limit = {
					has_idea = low_corruption
				}
				swap_ideas = {
					remove_idea = low_corruption
					add_idea = moderate_corruption
				}  
			}
			if = {
				limit = {
					has_idea = minor_corruption
				}
				swap_ideas = {
					remove_idea = minor_corruption
					add_idea = low_corruption
				}  
			}			
		}
	}

    option = {
        name = corruption.1.a
		add_stability = -0.05
		add_popularity = {
			ideology = ROOT
			popularity = -0.1
		}
    }

    option = {
        name = corruption.1.b
		add_political_power = -25
		add_stability = -0.1
		add_popularity = {
			ideology = ROOT
			popularity = -0.1
		}
    }
}
