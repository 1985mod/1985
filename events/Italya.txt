	title = italya.1.t
	desc = italya.1.desc
	hidden = yes

	trigger = {
		tag = ITA
		has_civil_war = no
		date > 1987.4.17
	}

	fire_only_once = yes

	option = {
		name = italya.1.a
		effect_tooltip = {
			set_politics = {

		}

			}

			}
		}
	}
}
