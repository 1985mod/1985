		allowed = { always = yes }
		

		cost = 25
		available = {
			has_political_power > 24

		}

		visible = { is_ai = yes }

		remove_effect = {
			add_to_variable = { var = pop_rate value = 1 }
			tax_effect = yes
			if = {
				limit = {
					check_variable = { pop_rate > 100 }

			}
		}

		ai_will_do = {
            base = 10
