on_actions = {
	on_monthly = {
		effect = {
			every_country = {
				set_temp_variable = { var = num_of_civ value = num_of_civilian_factories }
				multiply_temp_variable = { var = num_of_civ value = 15 }
				
				set_temp_variable = { var = num_of_mil value = num_of_military_factories }
				multiply_temp_variable = { var = num_of_mil value = 10 }

				set_temp_variable = { var = num_of_naval value = num_of_naval_factories }
				multiply_temp_variable = { var = num_of_naval value = 10 }

				set_variable = { var = real_gdp value = num_of_civ }
				add_to_variable = { var = real_gdp value = num_of_mil }
				add_to_variable = { var = real_gdp value = num_of_naval }

				add_to_array = { array = real_gdp_list value = real_gdp }

				set_variable = { var = old_real_gdp value = real_gdp_list^0 }
				
				if = {
					limit = { check_variable = { old_real_gdp < 1 } }
					set_variable = { var = old_real_gdp value = 1 }
				}

				set_variable = { var = gdp_growth_rate value = real_gdp }
				divide_variable = { var = gdp_growth_rate value = old_real_gdp }
				subtract_from_variable = { var = gdp_growth_rate value = 1 }
				multiply_variable = { var = gdp_growth_rate value = 100 }

				if = {
					limit = { check_variable = { real_gdp_list^3 > 0 } }
					remove_from_array = { array = real_gdp_list index = 0 }
				}

				set_variable = { var = gdp_per_capita value = real_gdp }

				set_variable = { var = saddam value = max_manpower }
				set_variable = { var = m_manpower_k value = 0.001 }
				multiply_variable = { var = m_manpower_k value = saddam }

				if = {
					limit = {
						max_manpower_k > 0
					}
					divide_variable = { var = gdp_per_capita value = m_manpower_k }
					multiply_variable = { var = gdp_per_capita value = 1000 }
				}
				else = {
					set_variable = { var = gdp_per_capita value = 0 }
				}

				set_variable = { var = civ_of_gdp value = num_of_civ }
				divide_variable = { var = civ_of_gdp value = real_gdp }
				add_to_variable = { var = civ_of_gdp value = 0.005 }

				set_variable = { var = mil_of_gdp value = num_of_mil }
				divide_variable = { var = mil_of_gdp value = real_gdp }
				add_to_variable = { var = mil_of_gdp value = 0.005 }

				set_variable = { var = naval_of_gdp value = num_of_naval }
				divide_variable = { var = naval_of_gdp value = real_gdp }
				add_to_variable = { var = naval_of_gdp value = 0.005 }

				set_variable = { var = real_gdp_divided_by_1000 value = real_gdp }
				divide_variable = { var = real_gdp_divided_by_1000 value = 1000 }

				set_variable = { var = old_real_gdp value = real_gdp_list^0 }

				set_variable = { var = gdp_growth_rate value = real_gdp }

				subtract_from_variable = { var = gdp_growth_rate value = old_real_gdp }

				divide_variable = { var = gdp_growth_rate value = old_real_gdp }

				multiply_variable = { var = gdp_growth_rate value = 100 }

				remove_from_array = {
					array = nominal_gdp_list
					index = 0
				}

				add_to_array = {
					array = nominal_gdp_list
					value = nominal_gdp
				}

				if = {
					limit = {
						check_variable = { var = inflation_rate value = -4 compare = less_than }
					}
					if = {
						limit = { NOT = { has_idea = acute_deflation } }
						remove_idea = deflation
						remove_idea = inflation 
						remove_idea = high_inflation 
						remove_idea = galloping_inflation 
						remove_idea = hyperinflation 
						add_ideas = acute_deflation
					}
				}
				else_if = {
					limit = {
						check_variable = { var = inflation_rate value = -4 compare = greater_than_or_equals }
						check_variable = { var = inflation_rate value = 0 compare = less_than }
					}
					if = {
						limit = { NOT = { has_idea = deflation } }
						remove_idea = acute_deflation 
						remove_idea = inflation 
						remove_idea = high_inflation 
						remove_idea = galloping_inflation 
						remove_idea = hyperinflation 
						add_ideas = deflation
					}
				}
				else_if = {
					limit = {
						check_variable = { var = inflation_rate value = 0 compare = greater_than_or_equals }
						check_variable = { var = inflation_rate value = 10 compare = less_than }
					}
					if = {
						limit = { NOT = { has_idea = inflation } }
						remove_idea = acute_deflation 
						remove_idea = deflation
						remove_idea = high_inflation 
						remove_idea = galloping_inflation 
						remove_idea = hyperinflation 
						add_ideas = inflation
					}
				}
				else_if = {
					limit = {
						check_variable = { var = inflation_rate value = 10 compare = greater_than_or_equals }
						check_variable = { var = inflation_rate value = 20 compare = less_than }
					}
					if = {
						limit = { NOT = { has_idea = high_inflation } }
						remove_idea = acute_deflation 
						remove_idea = deflation
						remove_idea = inflation 
						remove_idea = galloping_inflation 
						remove_idea = hyperinflation 
						add_ideas = high_inflation
					}
				}
				else_if = {
					limit = {
						check_variable = { var = inflation_rate value = 20 compare = greater_than_or_equals }
						check_variable = { var = inflation_rate value = 50 compare = less_than }
					}
					if = {
						limit = { NOT = { has_idea = galloping_inflation } }
						remove_idea = acute_deflation 
						remove_idea = deflation
						remove_idea = inflation 
						remove_idea = high_inflation 
						remove_idea = hyperinflation 
						add_ideas = galloping_inflation
					}
				}
				else_if = {
					limit = {
						check_variable = { var = inflation_rate value = 50 compare = greater_than_or_equals }
					}
					if = {
						limit = { NOT = { has_idea = hyperinflation } }
						remove_idea = acute_deflation 
						remove_idea = deflation
						remove_idea = inflation 
						remove_idea = high_inflation 
						remove_idea = galloping_inflation 
						add_ideas = hyperinflation
					}
				}

				set_variable = { var = pop_optimal value = 42.45 }
				set_temp_variable = { var = t value = 0.75 }
				multiply_temp_variable = { var = t value = tax_efficiency }
				set_temp_variable = { var = delitel value = 1.55 }
				subtract_from_temp_variable = { var = delitel value = t }
				divide_variable = { var = pop_optimal value = delitel }
				round_variable = pop_optimal

				set_variable = { var = corp_optimal value = 23.86 }
				set_temp_variable = { var = t value = 0.95 }
				multiply_temp_variable = { var = t value = tax_efficiency }
				set_temp_variable = { var = delitel value = 1.65 }
				subtract_from_temp_variable = { var = delitel value = t }
				divide_variable = { var = corp_optimal value = delitel }
				round_variable = corp_optimal

				set_variable = { var = vat_optimal value = 33.18 }
				set_temp_variable = { var = t value = 1.15 }
				multiply_temp_variable = { var = t value = tax_efficiency }
				set_temp_variable = { var = delitel value = 2.55 }
				subtract_from_temp_variable = { var = delitel value = t }
				divide_variable = { var = vat_optimal value = delitel }
				round_variable = vat_optimal
			}
		}
	}

	on_startup = {
		effect = {
			every_country = {

				if = {
					limit = {
						NOT = { has_dynamic_modifier = unemployment_dynamic_modifier }
					}
					add_dynamic_modifier = { modifier = unemployment_dynamic_modifier }
				}

				# Economy Law
				if = {
					limit = { has_idea = civilian_economy }
					set_variable = { var = army_multiplier value = 0.6 }
				}

				if = {
					limit = { has_idea = low_economic_mobilisation }
					set_variable = { var = army_multiplier value = 0.8 }
				}

				if = {
					limit = { has_idea = partial_economic_mobilisation }
					set_variable = { var = army_multiplier value = 1.0 }
				}

				if = {
					limit = { has_idea = war_economy }
					set_variable = { var = army_multiplier value = 1.3 }
				}

				if = {
					limit = { has_idea = tot_economic_mobilisation }
					set_variable = { var = army_multiplier value = 1.6 }
				}

				# Corruption Law
				if = { 
					limit = { has_idea = horrific_corruption }
					set_variable = { var = corruption_level value = 0.25 }
				}

				if = { 
					limit = { has_idea = significant_corruption }
					set_variable = { var = corruption_level value = 0.4 }
				}

				if = { 
					limit = { has_idea = moderate_corruption }
					set_variable = { var = corruption_level value = 0.6 }
				}

				if = { 
					limit = { has_idea = low_corruption }
					set_variable = { var = corruption_level value = 0.8 }
				}

				if = { 
					limit = { has_idea = minor_corruption }
					set_variable = { var = corruption_level value = 0.95 }
				}

				# Calculate GDP effect
				set_temp_variable = { var = num_of_civ value = num_of_civilian_factories }
				multiply_temp_variable = { var = num_of_civ value = 15 }
				
				set_temp_variable = { var = num_of_mil value = num_of_military_factories }
				multiply_temp_variable = { var = num_of_mil value = 10 }

				set_temp_variable = { var = num_of_naval value = num_of_naval_factories }
				multiply_temp_variable = { var = num_of_naval value = 10 }

				set_variable = { var = real_gdp value = num_of_civ }
				add_to_variable = { var = real_gdp value = num_of_mil }
				add_to_variable = { var = real_gdp value = num_of_naval }

				set_variable = { var = gdp_per_capita value = real_gdp }
				multiply_variable = { var = gdp_per_capita value = 10 }
				if = {
					limit = {
						max_manpower > 0
					}
					divide_variable = { var = gdp_per_capita value = max_manpower_k }
				}
				else = {
					set_variable = { var = gdp_per_capita value = 0 }
				}

				multiply_variable = { var = gdp_per_capita value = 100 }

				set_variable = { var = civ_of_gdp value = num_of_civ }
				divide_variable = { var = civ_of_gdp value = real_gdp }
				add_to_variable = { var = civ_of_gdp value = 0.005 }

				set_variable = { var = mil_of_gdp value = num_of_mil }
				divide_variable = { var = mil_of_gdp value = real_gdp }
				add_to_variable = { var = mil_of_gdp value = 0.005 }

				set_variable = { var = naval_of_gdp value = num_of_naval }
				divide_variable = { var = naval_of_gdp value = real_gdp }
				add_to_variable = { var = naval_of_gdp value = 0.005 }

				set_variable = { var = real_gdp_divided_by_1000 value = real_gdp }
				divide_variable = { var = real_gdp_divided_by_1000 value = 1000 }

				# Nominal GDP effect

				set_variable = { var = money_supply value = real_gdp }
				set_variable = { var = velocity value = 1 }

				set_variable = { var = nominal_gdp value = money_supply }
				multiply_variable = { var = nominal_gdp value = velocity }
				set_variable = { var = nominal_gdp_divided_by_1000 value = nominal_gdp }
				divide_variable = { var = nominal_gdp_divided_by_1000 value = 1000 }

				# Inflation effect
				if = { 
					limit = { NOT = { has_variable = inflation_rate } } 
					set_variable = { var = inflation_rate value = 0 } 
				}
				if = { 
					limit = { NOT = { has_variable = nominal_gdp } } 
					set_variable = { var = nominal_gdp value = real_gdp } 
				}

				if = {
					limit = {
						has_variable = real_gdp
						has_variable = nominal_gdp
						has_variable = inflation_rate
						check_variable = { var = real_gdp value = 0 compare = greater_than }
					}
					
					if = {
						limit = {
							NOT = { has_variable = old_inflation }
						}
						set_variable = { var = old_inflation value = 0 }
					}

					set_temp_variable = { var = instant_inflation value = nominal_gdp }
					divide_temp_variable = { var = instant_inflation value = real_gdp }
					subtract_from_temp_variable = { var = instant_inflation value = 1 }
					multiply_temp_variable = { var = instant_inflation value = 100 }
					
					set_temp_variable = { var = weighted_old value = old_inflation }
					multiply_temp_variable = { var = weighted_old value = 0.8 }

					set_temp_variable = { var = weighted_instant value = instant_inflation }
					multiply_temp_variable = { var = weighted_instant value = 0.2 }
					
					set_variable = { var = inflation_rate value = weighted_old }
					add_to_variable = { var = inflation_rate value = weighted_instant }

					set_variable = { var = old_inflation value = inflation_rate }

					if = {
						limit = { check_variable = { debt_to_gdp > 150 } }
						
						set_temp_variable = { var = inflation_growth_temp value = debt_to_gdp }
						subtract_from_temp_variable = { var = inflation_growth_temp value = 150 }
						
						divide_temp_variable = { var = inflation_growth_temp value = 10 }
						multiply_temp_variable = { var = inflation_growth_temp value = 1.5 }
						
						add_to_variable = { var = inflation_rate value = inflation_growth_temp }
					}
				}

				# Corruption effect

				if = {
					limit = {
						NOT = { has_variable = corruption_level }
					}
					set_variable = { var = corruption_level value = 0.4 }
				}

				if = { 
					limit = {
						has_variable = corruption_level
						has_idea = horrific_corruption
					}
					set_variable = { var = corruption_level value = 0.25 }
				}

				if = { 
					limit = {
						has_variable = corruption_level
						has_idea = significant_corruption
					}
					set_variable = { var = corruption_level value = 0.4 }
				}

				if = { 
					limit = {
						has_variable = corruption_level
						has_idea = moderate_corruption
					}
					set_variable = { var = corruption_level value = 0.6 }
				}

				if = { 
					limit = {
						has_variable = corruption_level
						has_idea = low_corruption
					}
					set_variable = { var = corruption_level value = 0.8 }
				}

				if = { 
					limit = {
						has_variable = corruption_level
						has_idea = minor_corruption
					}
					set_variable = { var = corruption_level value = 0.95 }
				}

				if = { 
					limit = {
						has_government = communism
					}

					set_variable = { var = corruption_income_mod value = 0.5 }
					multiply_variable = { var = corruption_income_mod value = corruption_level }
					add_to_variable = { var = corruption_income_mod value = 0.5 }
				}
				else = {
					set_variable = { var = corruption_income_mod value = corruption_level }
				}

				set_variable = { var = tax_efficiency value = 0.6 }
				multiply_variable = { var = tax_efficiency value = gdp_per_capita }
				
				set_temp_variable = { var = tax_efficiency_temp value = gdp_per_capita }
				add_to_temp_variable = { var = tax_efficiency_temp value = 3000 }
				
				divide_variable = { var = tax_efficiency value = tax_efficiency_temp }
				add_to_variable = { var = tax_efficiency value = 0.35 }

				multiply_variable = { var = tax_efficiency value = corruption_income_mod }

				# Set default taxes

				set_variable = { var = pop_rate value = 20 }
				set_variable = { var = corp_rate value = 15 }
				set_variable = { var = vat_rate value = 10 }
				set_variable = { var = interest_rate value = 0 }

				# Tax effect

				set_variable = { var = population_efficiency value = 1 }

				set_temp_variable = { var = a value = tax_efficiency }
				multiply_temp_variable = { var = a value = 0.75 }

				set_temp_variable = { var = b value = 1.55 }
				subtract_from_temp_variable = { var = b value = a }
				multiply_temp_variable = { var = b value = 1.36 }
				multiply_temp_variable = { var = b value = pop_rate }
				divide_temp_variable = { var = b value = 100 }
				multiply_temp_variable = { var = b value = b }

				subtract_from_variable = { var = population_efficiency value = b }
				multiply_variable = { var = population_efficiency value = tax_efficiency }

				if = {
					limit = { check_variable = { population_efficiency < 0 } }
					set_variable = { var = population_efficiency value = 0 }
				}

				if = {
					limit = { NOT = { has_variable = k_pop } }
					if = {
						limit = { has_government = communism }
						set_variable = { var = k_pop value = 2.25 }
					}
					else = {
						set_variable = { var = k_pop value = 3.75 }
					}
				}
			
				set_variable = { var = population_tax value = k_pop }
				multiply_variable = { var = population_tax value = nominal_gdp }
				divide_variable = { var = population_tax value = 100 }
				multiply_variable = { var = population_tax value = pop_rate }
				multiply_variable = { var = population_tax value = population_efficiency }
				multiply_variable = { var = population_tax value = 7 }
				divide_variable = { var = population_tax value = 365 }

				set_variable = { var = corporate_efficiency value = 1 }

				set_temp_variable = { var = a value = tax_efficiency }
				multiply_temp_variable = { var = a value = 0.95 }

				set_temp_variable = { var = b value = 1.65 }
				subtract_from_temp_variable = { var = b value = a }
				multiply_temp_variable = { var = b value = 2.42 }
				multiply_temp_variable = { var = b value = corp_rate }
				divide_temp_variable = { var = b value = 100 }
				multiply_temp_variable = { var = b value = b }

				subtract_from_variable = { var = corporate_efficiency value = b }
				multiply_variable = { var = corporate_efficiency value = tax_efficiency }

				if = {
					limit = { check_variable = { corporate_efficiency < 0 } }
					set_variable = { var = corporate_efficiency value = 0 }
				}

				if = {
					limit = { NOT = { has_variable = k_corp } }
					if = {
						limit = { has_government = communism }
						set_variable = { var = k_corp value = 4.0 }
					}
					else = {
						set_variable = { var = k_corp value = 1.75 }
					}
				}
			
				set_variable = { var = corporate_tax value = k_corp }
				multiply_variable = { var = corporate_tax value = nominal_gdp }
				divide_variable = { var = corporate_tax value = 100 }
				multiply_variable = { var = corporate_tax value = corp_rate }
				multiply_variable = { var = corporate_tax value = corporate_efficiency }
				multiply_variable = { var = corporate_tax value = 7 }
				divide_variable = { var = corporate_tax value = 365 }

				set_variable = { var = vat_efficiency value = 1 }

				set_temp_variable = { var = a value = tax_efficiency }
				multiply_temp_variable = { var = a value = 1.15 }

				set_temp_variable = { var = b value = 2.55 }
				subtract_from_temp_variable = { var = b value = a }
				multiply_temp_variable = { var = b value = 1.74 }
				multiply_temp_variable = { var = b value = vat_rate }
				divide_temp_variable = { var = b value = 100 }
				multiply_temp_variable = { var = b value = b }

				subtract_from_variable = { var = vat_efficiency value = b }
				multiply_variable = { var = vat_efficiency value = tax_efficiency }

				if = {
					limit = { check_variable = { vat_efficiency < 0 } }
					set_variable = { var = vat_efficiency value = 0 }
				}

				if = {
					limit = { NOT = { has_variable = k_vat } }
					if = {
						limit = { has_government = communism }
						set_variable = { var = k_vat value = 2.5 }
					}
					else = {
						set_variable = { var = k_vat value = 1.5 }
					}
				}
			
				set_variable = { var = vat_tax value = k_vat }
				multiply_variable = { var = vat_tax value = nominal_gdp }
				divide_variable = { var = vat_tax value = 100 }
				multiply_variable = { var = vat_tax value = vat_rate }
				multiply_variable = { var = vat_tax value = vat_efficiency }
				multiply_variable = { var = vat_tax value = 7 }
				divide_variable = { var = vat_tax value = 365 }

				set_variable = { var = export_oil value = resource_imported@oil }
				multiply_variable = { var = export_oil value = 0.04 }

				set_variable = { var = export_tungsten value = resource_imported@tungsten }
				multiply_variable = { var = export_tungsten value = 0.03 }

				set_variable = { var = export_aluminum value = resource_imported@aluminum }
				multiply_variable = { var = export_aluminum value = 0.3 }

				set_variable = { var = export_steel value = resource_imported@steel }
				multiply_variable = { var = export_steel value = 0.034 }

				set_variable = { var = export_chromium value = resource_imported@chromium }
				multiply_variable = { var = export_chromium value = 0.037 }

				set_variable = { var = export_rubber value = resource_imported@rubber }
				multiply_variable = { var = export_rubber value = 0.036 }

				set_variable = { var = export_coal value = resource_imported@coal }
				multiply_variable = { var = export_coal value = 0.02 }

				set_variable = { var = resource_revenue value = export_oil }
				add_to_variable = { var = resource_revenue value = export_tungsten }
				add_to_variable = { var = resource_revenue value = export_aluminum }
				add_to_variable = { var = resource_revenue value = export_steel }
				add_to_variable = { var = resource_revenue value = export_chromium }
				add_to_variable = { var = resource_revenue value = export_rubber }
				add_to_variable = { var = resource_revenue value = export_coal }

				multiply_variable = { var = resource_revenue value = corruption_income_mod }

				clamp_variable = {
					var = population_tax
					min = 0
				}

				clamp_variable = {
					var = corporate_tax
					min = 0
				}

				clamp_variable = {
					var = vat_tax
					min = 0
				}

				clamp_variable = {
					var = resource_revenue
					min = 0
				}

				set_variable = { var = income_budget value = population_tax }
				add_to_variable = { var = income_budget value = corporate_tax }
				add_to_variable = { var = income_budget value = vat_tax }
				add_to_variable = { var = income_budget value = resource_revenue }
				add_to_variable = { var = income_budget value = modifier@income_growth_factor }

				set_temp_variable = { var = A value = max_manpower_k }
				multiply_temp_variable = { var = A value = 0.04 }
				multiply_temp_variable = { var = A value = gdp_per_capita }
				divide_temp_variable = { var = A value = 1000000 }

				set_temp_variable = { var = B value = nominal_gdp }
				multiply_temp_variable = { var = B value = 0.0001 }
				set_temp_variable = { var = B_k value = 1 }
				subtract_from_temp_variable = { var = B_k value = corruption_level }
				multiply_temp_variable = { var = B_k value = 0.5 }
				add_to_temp_variable = { var = B_k value = 1 }
