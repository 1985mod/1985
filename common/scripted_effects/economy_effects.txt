calculate_gdp_effect = { 
    every_country = { 
        set_variable = { var = num_of_civ value = num_of_civilian_factories }
        multiply_variable = { var = num_of_civ value = 15 }
        
        set_variable = { var = num_of_mil value = num_of_military_factories }
        multiply_variable = { var = num_of_mil value = 10 }

        set_variable = { var = num_of_naval value = num_of_naval_factories }
        multiply_variable = { var = num_of_naval value = 10 }

        set_variable = { var = country_gdp value = num_of_civ }
        add_to_variable = { var = country_gdp value = num_of_mil }
        add_to_variable = { var = country_gdp value = num_of_naval }

        set_variable = { var = gdp_per_capita value = country_gdp }
        multiply_variable = { var = gdp_per_capita value = 10000 }
        if = {
            limit = {
                max_manpower > 0
            }
            divide_variable = { var = gdp_per_capita value = max_manpower }
        }
        multiply_variable = { var = gdp_per_capita value = 100 }
    }
}

calculate_nominal_gdp_effect = { 
    every_country = { 
        if = {
            limit = { NOT = { has_variable = money_supply } }
            set_variable = { var = money_supply value = country_gdp }
        }
        if = {
            limit = { check_variable = { var = velocity value = 0 compare = less_than } }
            set_variable = { var = velocity value = 0 }
        }
        set_variable = { var = nominal_gdp value = money_supply }
        multiply_variable = { var = nominal_gdp value = velocity }
    }
}

calculate_gdp_growth = {
    every_country = {
        if = {
            limit = { 
                NOT = { has_variable = months_passed_timer } 
            }
            set_variable = { var = months_passed_timer value = 0 }
            set_variable = { var = old_gdp value = country_gdp }
        }

        add_to_variable = { var = months_passed_timer value = 1 }

        if = {
            limit = {
                check_variable = { months_passed_timer > 2 }
            }
        }
            
        if = {
            limit = {
                check_variable = { var = old_gdp value = 0 compare = greater_than }
            }
            set_temp_variable = { var = gdp_growth_calculate value = country_gdp }
            subtract_from_temp_variable = { var = gdp_growth_calculate value = old_gdp }
            multiply_temp_variable = { var = gdp_growth_calculate value = 100 }
            divide_temp_variable = { var = gdp_growth_calculate value = old_gdp }
            set_variable = { var = gdp_growth_rate value = gdp_growth_calculate }
        }   
            
        else = {
            set_variable = { var = gdp_growth_rate value = 0 }
            set_variable = { var = old_gdp value = country_gdp }
            clear_variable = months_passed_timer
            set_variable = { var = months_passed_timer value = 0 }
        }
    }
}

inflation_effect = {
    every_country = {   
        if = { 
            limit = { NOT = { has_variable = inflation_rate } } 
            set_variable = { var = inflation_rate value = 0 } 
        }
        if = { 
            limit = { NOT = { has_variable = nominal_gdp } } 
            set_variable = { var = nominal_gdp value = country_gdp } 
        }

        if = {
            limit = {
                has_variable = country_gdp
                has_variable = nominal_gdp
                has_variable = inflation_rate
                check_variable = { var = country_gdp value = 0 compare = greater_than }
            }
            
            if = {
                limit = {
                    NOT = { has_variable = old_inflation }
                }
                set_variable = { var = old_inflation value = 0 }
            }
            
            set_temp_variable = { var = instant_inflation value = nominal_gdp }
            divide_temp_variable = { var = instant_inflation value = country_gdp }
            subtract_from_temp_variable = { var = instant_inflation value = 1 }
            multiply_temp_variable = { var = instant_inflation value = 100 }
            
            set_temp_variable = { var = weighted_old value = old_inflation }
            multiply_temp_variable = { var = weighted_old value = 0.8 }

            set_temp_variable = { var = weighted_instant value = instant_inflation }
            multiply_temp_variable = { var = weighted_instant value = 0.2 }
            
            set_variable = { var = inflation_rate value = weighted_old }
            add_to_variable = { var = inflation_rate value = weighted_instant }

            set_variable = { var = old_inflation value = inflation_rate }
        }
    }
}

corruption_effect = {
    if = {
        limit = {
            NOT = { has_variable = corruption_level }
        }
        set_variable = { var = corruption_level value = 0.4 }
    }

    if = { 
        limit = {
            has_variable = corruption_level
            has_idea = horrific_corruption
        }
        set_variable = { var = corruption_level value = 0.25 }
    }

    if = { 
        limit = {
            has_variable = corruption_level
            has_idea = significant_corruption
        }
        set_variable = { var = corruption_level value = 0.4 }
    }

    if = { 
        limit = {
            has_variable = corruption_level
            has_idea = moderate_corruption
        }
        set_variable = { var = corruption_level value = 0.6 }
    }

    if = { 
        limit = {
            has_variable = corruption_level
            has_idea = low_corruption
        }
        set_variable = { var = corruption_level value = 0.8 }
    }

    if = { 
        limit = {
            has_variable = corruption_level
            has_idea = minor_corruption
        }
        set_variable = { var = corruption_level value = 0.95 }
    }

    if = { 
        limit = {
            has_government = communism
            NOT = { has_variable = corruption_income_mod }
        }

        set_variable = { var = corruption_income_mod value = 0.5 }
        multiply_variable = { var = corruption_income_mod value = corruption_level }
        add_to_variable = { var = corruption_income_mod value = 0.5 }
    }

    if = {
        limit = {
            NOT = { has_variable = corruption_income_mod }
        }
        set_variable = { var = corruption_income_mod value = corruption_level }
    }

    if = {
        limit = {
            NOT = { has_variable = tax_effiency }
        }
        set_variable = { var = tax_effiency value = 0.6 }
        multiply_variable = { var = tax_effiency value = gdp_per_capita }
        
        set_temp_variable = { var = tax_effiency_temp value = gdp_per_capita }
        add_to_temp_variable = { var = tax_effiency_temp value = 3000 }
        
        divide_variable = { var = tax_effiency value = tax_effiency_temp }
        add_to_variable = { var = tax_effiency value = 0.35 }

        multiply_variable = { var = tax_effiency value = corruption_income_mod }
    }
}

velocity_effect = {
    every_country = {
        if = {
            limit = { NOT = { has_variable = interest_rate } }
            set_variable = { var = interest_rate value = 0 }
        }
        if = {
            limit = {
                has_variable = interest_rate
                has_variable = inflation_rate
            }
            set_temp_variable = { var = i value = interest_rate }
            divide_temp_variable = { var = i value = 30 }

            set_temp_variable = { var = f value = inflation_rate }
            divide_temp_variable = { var = f = value = 50 }

            set_temp_variable = { var = putin value = 1 }
            set_temp_variable = { var = zelensky value = 1 }
            add_to_temp_variable = { var = zelensky value = i }
            divide_temp_variable = { var = putin value = zelensky }

            set_temp_variable = { var = admiral_biden value = f }

            if = {
                limit = { check_variable = { var = inflation_rate value = 0 compare = greater_than } }
                multiply_temp_variable = { var = admiral_biden value = 2 }
            }
            else = {
                multiply_temp_variable = { var = admiral_biden value = -1.1 }
            }

            add_to_temp_variable = { var = admiral_biden value = 100 }
            divide_temp_variable = { var = f value = admiral_biden }
            add_to_temp_variable = { var = putin value = f }

            set_variable = { var = velocity value = putin }
        }
    }
}
