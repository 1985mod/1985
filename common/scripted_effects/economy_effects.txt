calculate_gdp_effect = { 
    every_country = { 
        set_variable = { var = num_of_civ value = num_of_civilian_factories }
        multiply_variable = { var = num_of_civ value = 15 }
        
        set_variable = { var = num_of_mil value = num_of_military_factories }
        multiply_variable = { var = num_of_mil value = 10 }

        set_variable = { var = num_of_naval value = num_of_naval_factories }
        multiply_variable = { var = num_of_naval value = 10 }

        set_variable = { var = country_gdp value = num_of_civ }
        add_to_variable = { var = country_gdp value = num_of_mil }
        add_to_variable = { var = country_gdp value = num_of_naval }

        set_variable = { var = gdp_per_capita value = country_gdp }
        multiply_variable = { var = gdp_per_capita value = 10000 }
        if = {
            limit = {
                max_manpower > 0
            }
            divide_variable = { var = gdp_per_capita value = max_manpower }
        }
        multiply_variable = { var = gdp_per_capita value = 100 }

		if = {
			limit = {
				check_variable = { var = old_gdp value = 0 compare = greater_than }
			}
			set_variable = { var = gdp_growth_calculate value = country_gdp }
			subtract_from_variable  = { var = gdp_growth_calculate value = old_gdp }
			multiply_variable = { var = gdp_growth_calculate value = 100 }
			divide_variable = { var = gdp_growth_calculate value = old_gdp }

			set_variable = { var = gdp_growth_rate value = gdp_growth_calculate }
		}

		else = {
			set_variable = { var = gdp_growth_rate value = 0 }
		}

        set_variable = { var = old_gdp value = country_gdp }
    }
}
