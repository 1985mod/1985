calculate_gdp_effect = { 
    every_country = {
        set_temp_variable = { var = num_of_civ value = num_of_civilian_factories }
        multiply_temp_variable = { var = num_of_civ value = 15 }
        
        set_temp_variable = { var = num_of_mil value = num_of_military_factories }
        multiply_temp_variable = { var = num_of_mil value = 10 }

        set_temp_variable = { var = num_of_naval value = num_of_naval_factories }
        multiply_temp_variable = { var = num_of_naval value = 10 }

        set_variable = { var = country_gdp value = num_of_civ }
        add_to_variable = { var = country_gdp value = num_of_mil }
        add_to_variable = { var = country_gdp value = num_of_naval }

        if = {
            limit = {
                check_variable = { k_unemployment_rate < 0 }
            }
            set_temp_variable = { var = temp_k_unemployment_rate value = k_unemployment_rate }
            add_to_temp_variable =  { var = temp_k_unemployment_rate value = 1 }
            multiply_variable = { var = country_gdp value = temp_k_unemployment_rate }
        }

        set_variable = { var = gdp_per_capita value = country_gdp }
        multiply_variable = { var = gdp_per_capita value = 10000 }
        if = {
            limit = {
                max_manpower > 0
            }
            divide_variable = { var = gdp_per_capita value = max_manpower }
        }
        else = {
            set_variable = { var = gdp_per_capita value = 0 }
        }

        multiply_variable = { var = gdp_per_capita value = 100 }

        set_variable = { var = civ_of_gdp value = num_of_civ }
        divide_variable = { var = civ_of_gdp value = country_gdp }
        add_to_variable = { var = civ_of_gdp value = 0.005 }

        set_variable = { var = mil_of_gdp value = num_of_mil }
        divide_variable = { var = mil_of_gdp value = country_gdp }
        add_to_variable = { var = mil_of_gdp value = 0.005 }

        set_variable = { var = naval_of_gdp value = num_of_naval }
        divide_variable = { var = naval_of_gdp value = country_gdp }
        add_to_variable = { var = naval_of_gdp value = 0.005 }

        set_variable = { var = country_gdp_divided_by_1000 value = country_gdp }
        divide_variable = { var = country_gdp_divided_by_1000 value = 1000 }
    }
}

calculate_nominal_gdp_effect = { 
    every_country = { 
        if = {
            limit = { NOT = { has_variable = money_supply } }
            set_variable = { var = money_supply value = country_gdp }
        }
        if = {
            limit = { check_variable = { var = velocity value = 0 compare = less_than } }
            set_variable = { var = velocity value = 0 }
        }
        set_variable = { var = nominal_gdp value = money_supply }
        multiply_variable = { var = nominal_gdp value = velocity }
        set_variable = { var = nominal_gdp_divided_by_1000 value = nominal_gdp }
        divide_variable = { var = nominal_gdp_divided_by_1000 value = 1000 }
    }
}

calculate_gdp_growth = {
    every_country = {
        set_variable = { var = old_gdp_2 value = old_gdp_1 }
        set_variable = { var = old_gdp_1 value = old_gdp_0 }
        set_variable = { var = old_gdp_0 value = country_gdp }

        if = {
            limit = {
                check_variable = { var = old_gdp_2 value = 0 compare = greater_than }
            }
            set_temp_variable = { var = gdp_growth_calculate value = country_gdp }
            subtract_from_temp_variable = { var = gdp_growth_calculate value = old_gdp_2 }
            multiply_temp_variable = { var = gdp_growth_calculate value = 100 }
            divide_temp_variable = { var = gdp_growth_calculate value = old_gdp_2 }
            set_variable = { var = gdp_growth_rate value = gdp_growth_calculate }
        }   
            
        else = {
            set_variable = { var = gdp_growth_rate value = 0 }
        }
    }
}

inflation_effect = {
    every_country = {   
        if = { 
            limit = { NOT = { has_variable = inflation_rate } } 
            set_variable = { var = inflation_rate value = 0 } 
        }
        if = { 
            limit = { NOT = { has_variable = nominal_gdp } } 
            set_variable = { var = nominal_gdp value = country_gdp } 
        }

        if = {
            limit = {
                has_variable = country_gdp
                has_variable = nominal_gdp
                has_variable = inflation_rate
                check_variable = { var = country_gdp value = 0 compare = greater_than }
            }
            
            if = {
                limit = {
                    NOT = { has_variable = old_inflation }
                }
                set_variable = { var = old_inflation value = 0 }
            }

            set_temp_variable = { var = instant_inflation value = nominal_gdp }
            divide_temp_variable = { var = instant_inflation value = country_gdp }
            subtract_from_temp_variable = { var = instant_inflation value = 1 }
            multiply_temp_variable = { var = instant_inflation value = 100 }
            
            set_temp_variable = { var = weighted_old value = old_inflation }
            multiply_temp_variable = { var = weighted_old value = 0.8 }

            set_temp_variable = { var = weighted_instant value = instant_inflation }
            multiply_temp_variable = { var = weighted_instant value = 0.2 }
            
            set_variable = { var = inflation_rate value = weighted_old }
            add_to_variable = { var = inflation_rate value = weighted_instant }

            set_variable = { var = old_inflation value = inflation_rate }
        }
    }
}

velocity_effect = {
    every_country = {
        if = {
            limit = { NOT = { has_variable = interest_rate } }
            set_variable = { var = interest_rate value = 0 }
        }
        if = {
            limit = {
                has_variable = interest_rate
                has_variable = inflation_rate
            }
            set_temp_variable = { var = rate_divisor value = 80 }
            if = {
                limit = { check_variable = { var = interest_rate compare = less_than } }
                set_temp_variable = { var = rate_divisor value = 10 }
            }

            set_temp_variable = { var = normalized_rate value = interest_rate }
            divide_temp_variable = { var = normalized_rate value = 80 }

            set_temp_variable = { var = base_velocity value = 1 }
            
            set_temp_variable = { var = inflation_impact value = inflation_rate }
            divide_temp_variable = { var = inflation_impact value = 20 }

            set_temp_variable = { var = interest_rate_impact value = interest_rate }
            divide_temp_variable = { var = interest_rate_impact value = 80 }

            

            if = { 
                limit = { check_variable = { var = base_velocity value = 0.5 compare = less_than } }
