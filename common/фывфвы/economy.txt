scripted_gui = {
	economy_hide = {
		context_type = player_context

		window_name = "economy_hide_button"

		visible = {
			ROOT = { NOT = { has_country_flag = economy_hide } }
		}

		effects = {
			economy_hide_bg_click = {
				ROOT = { set_country_flag = economy_hide }
			}
		}
	}
	economy_show = {
		context_type = player_context

		window_name = "economy_show_button"

		visible = {
			ROOT = { has_country_flag = economy_hide }
		}

		effects = {
			economy_show_bg_click = {
				ROOT = { clr_country_flag = economy_hide }
			}
		}
	}
	take_duty = {
		context_type = player_context

		window_name = "taxes_container"

		visible = {
			ROOT = { NOT = { has_country_flag = hide_money } }
		}

		effects = {
			#Take debt
			duty_bg_click = {
				if = {
					add_to_variable = { coffers = 1 }
					add_to_variable = { duty = 1.01 }
					ingame_update_setup = yes
				}
			}
		}
		debt_bg_control_click = {
			if = {
				add_to_variable = { coffers = 10 }
				add_to_variable = { duty = 10.1 }
				ingame_update_setup = yes
			}
		}
		debt_bg_shift_click = {
			if = {
				add_to_variable = { coffers = 100 }
				add_to_variable = { duty = 101 }
				ingame_update_setup = yes
			}
		}
		debt_bg_alt_click = {
			if = {
				add_to_variable = { coffers = 1000 }
				add_to_variable = { duty = 1010 }
				ingame_update_setup = yes
			}
		}
	}
}
