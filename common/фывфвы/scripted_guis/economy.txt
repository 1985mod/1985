scripted_gui = {
	economy_hide = {
		context_type = player_context

		window_name = "economy_hide_button"

		visible = {
			ROOT = { NOT = { has_country_flag = economy_hide } }
		}

		effects = {
			economy_hide_bg_click = {
				ROOT = { set_country_flag = economy_hide }
			}
		}
	}
	economy_show = {
		context_type = player_context

		window_name = "economy_show_button"

		visible = {
			ROOT = { has_country_flag = economy_hide }
		}

		effects = {
			economy_show_bg_click = {
				ROOT = { clr_country_flag = economy_hide }
			}
		}
	}
}
