SOV_invade_infantry_in_afghanistan = {
    allowed = {
        original_tag = SOV
    }
    enable = {
        country_exists = AFN
        AFS = { has_war_with = AFN }
    }
    abort = {
        NOT = { AFN = { has_war_with = AFS } }
    }
    abort_when_not_enabled = yes
    ai_strategy = {
        type = send_volunteers_desire
        id = AFN
        value = 1000
    }
}

global_lend_lease_and_volunteers = {
    allowed = {
        always = yes # Работает для ВСЕХ стран в игре
    }
    
    enable = {
        any_country = { is_at_war = yes } # Активируется, когда идет любая война
    }
    
    abort = {
        always = no # Никогда не отключается
    }
    
    # Из-за того, что здесь ВООБЩЕ НЕТ строки "id", 
    # движок HOI4 автоматически применяет этот вес ко ВСЕМ остальным странам мира!
    ai_strategy = {
        type = send_lend_lease_desire
        value = 20000
    }
    
    ai_strategy = {
        type = send_volunteers_desire
        value = 30000
    }
}
