scripted_gui = {
    economy = {
        context_type = player_context
        window_name = "economy"

        visible = {
            has_country_flag = economy_gui
        }

        effects = {
            button_4_click = {
                clr_country_flag = economy_gui
            }
        }

        effects = {
            interest_rate_increase_click = {
                add_to_variable = { var = interest_rate value = 1 }
                velocity_effect = yes
                if = {
                    limit = { 
                        check_variable = { interest_rate > 100 } 
                    }
                    set_variable = { var = interest_rate value = 100 }
                }
                else = {
                    add_political_power = -75
                }
            }
            interest_rate_decrease_click = {
                subtract_from_variable = { var = interest_rate value = 1 }
                velocity_effect = yes
                if = {
                    limit = { 
                        check_variable = { interest_rate < -5 } 
                    }
                    set_variable = { var = interest_rate value = -5 }
                }
                else = {
                    add_political_power = -75
                }
            }
            population_taxes_increase_click = {
                add_to_variable = { var = pop_rate value = 1 }
                tax_effect = yes
                if = {
                    limit = {
                        check_variable = { pop_rate > 100 } 
                    }
                    set_variable = { var = pop_rate value = 100 }
                }
                else = {
                    add_political_power = -50
                }
            }
            population_taxes_decrease_click = {
                subtract_from_variable = { var = pop_rate value = 1 }
                tax_effect = yes
                if = {
                    limit = {
                        check_variable = { pop_rate < 0 } 
                    }
                    set_variable = { var = pop_rate value = 0 }
                }
                else = {
                    add_political_power = -50
                }
            }
            corporate_rate_increase_click = {
                add_to_variable = { var = corp_rate value = 1 }
                tax_effect = yes
                if = {
                    limit = {
                        check_variable = { corp_rate > 100 } 
                    }
                    set_variable = { var = corp_rate value = 100 }
                }
                else = {
                    add_political_power = -50
                }
            }
            corporate_rate_decrease_click = {
                subtract_from_variable = { var = corp_rate value = 1 }
                tax_effect = yes
                if = {
                    limit = {
                        check_variable = { corp_rate < 0 } 
                    }
                    set_variable = { var = corp_rate value = 0 }
                }
                else = {
                    add_political_power = -50
                }
            }
            vat_rate_increase_click = {
                add_to_variable = { var = vat_rate value = 1 }
                tax_effect = yes
                if = {
                    limit = {
                        check_variable = { vat_rate > 100 } 
                    }
                    set_variable = { var = vat_rate value = 100 }
                }
                else = {
                    add_political_power = -50
                }
            }
            vat_rate_decrease_click = {
                subtract_from_variable = { var = vat_rate value = 1 }
                tax_effect = yes
                if = {
                    limit = {
                        check_variable = { vat_rate < 0 } 
                    }
                    set_variable = { var = vat_rate value = 0 }
                }
                else = {
                    add_political_power = -50
                }
            }
        }

        triggers = {
            button_4_click_enabled = {
                always = yes
            }
            interest_rate_increase_click_enabled = {
                ROOT = {
                    has_political_power > 74
                }
            }
            interest_rate_decrease_click_enabled = {
                ROOT = {
                    has_political_power > 74
                }
            }
            population_taxes_increase_click_enabled = {
                ROOT = {
                    has_political_power > 49
                }
            }
            population_taxes_decrease_click_enabled = {
                ROOT = {
                    has_political_power > 49
                }
            }
            corporate_rate_increase_click_enabled = {
                ROOT = {
                    has_political_power > 49
                }
            }
            corporate_rate_decrease_click_enabled = {
                ROOT = {
                    has_political_power > 49
                }
            }
            vat_rate_increase_click_enabled = {
                ROOT = {
                    has_political_power > 49
                }
            }
            vat_rate_decrease_click_enabled = {
                ROOT = {
                    has_political_power > 49
                }
            }            
        }
    }
}
