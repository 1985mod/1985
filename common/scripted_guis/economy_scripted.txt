scripted_gui = {
    debt = { 
        context_type = player_context
        window_name = "topbar_economy"

        visible = {
            always = yes 
        }
        effects = {
            debt_bg_click = {
                add_to_variable = { var = current_debt value = 1 }
                add_to_variable = { var = treasury value = 1 }
            }
            debt_bg_right_click = {
                subtract_from_variable = { var = current_debt value = 1 }
                subtract_from_variable = { var = treasury value = 1 }
                if = {
                    limit = {
                        check_variable = { current_debt < 0 } 
                    }
                    set_variable = { var = current_debt value = 0 }
                }
            }
            debt_bg_control_right_click = {
                subtract_from_variable = { var = current_debt value = 10 }
                subtract_from_variable = { var = treasury value = 10 }
            }
            debt_bg_control_click = {
                add_to_variable = { var = current_debt value = 10 }
                add_to_variable = { var = treasury value = 10 }
                if = {
                    limit = {
                        check_variable = { current_debt < 0 } 
                    }
                    set_variable = { var = current_debt value = 0 }
                }
            }
            debt_bg_shift_click = {
                add_to_variable = { var = current_debt value = 100 }
                add_to_variable = { var = treasury value = 100 }
            }
            debt_bg_shift_right_click = {
                subtract_from_variable = { var = current_debt value = 100 }
                subtract_from_variable = { var = treasury value = 100 }
                if = {
                    limit = {
                        check_variable = { current_debt < 0 } 
                    }
                    set_variable = { var = current_debt value = 0 }
                }
            }
        }
        triggers = {
            debt_bg_right_click_enabled = {
                ROOT = {
                    check_variable = { treasury > 1 }
                }
            }
            debt_bg_control_right_click_enabled = {
                ROOT = {
                    check_variable = { treasury > 10 }
                }
            }
            debt_bg_shift_right_click_enabled = {
                ROOT = {
                    check_variable = { treasury > 100 }
                }
            }
        }
    }
}
